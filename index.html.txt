        function calculateCosts() {
            const salaryValue = parseFloat(salary.value) || 0;
            const months = parseInt(employmentMonths.value) || 0;
            const weeks = parseInt(recruitmentWeeks.value) || 0;
            
            // Update total label with position name
            const positionName = position.value.trim();
            if (positionName) {
                totalLabelEl.textContent = `Koszt Błędnej Rekrutacji: ${positionName}`;
            } else {
                totalLabelEl.textContent = 'Całkowity Koszt Błędnej Rekrutacji';
            }
            
            // 1. Koszt procesu rekrutacyjnego
            const recruitmentProcessCost = 
                parseFloat(adCost.value || 0) +
                (parseFloat(recruiterHours.value || 0) * parseFloat(recruiterRate.value || 0)) +
                (parseFloat(managerHours.value || 0) * parseFloat(managerRate.value || 0));

            // 2. Wypłacone wynagrodzenie + ZUS pracodawcy (20.48% - dane 2026)
            const zusEmployer = 0.2048; // 9.76% + 6.5% + 1.67% + 2.45% + 0.1%
            const totalSalaryCost = salaryValue * months * (1 + zusEmployer);

            // 3. Koszty onboardingu
            const onboardingHours = 30; // Średni czas wdrożenia przez rekrutera/HR
            const mentorHours = 40; // Średni czas mentora/managera
            const onboardingTotal = 
                (onboardingHours * parseFloat(recruiterRate.value || 0)) +
                (mentorHours * parseFloat(managerRate.value || 0)) +
                parseFloat(trainingCost.value || 0);

            // 4. Utracona produktywność
            const productivityPercent = parseFloat(productivityLoss.value || 0) / 100;
            const productivityCostTotal = salaryValue * months * productivityPercent;

            // 5. Koszt ponownej rekrutacji (120% pierwszej - pilność)
            const repeatRecruitmentCost = recruitmentProcessCost * 1.2;

            // 6. Wpływ na zespół
            const teamImpactPercent = parseFloat(teamImpact.value || 0) / 100;
            const avgTeamSalary = salaryValue * 0.6; // Założenie: średnia pensja zespołu to 60% pensji tej osoby
            const teamProductivityLoss = avgTeamSalary * parseInt(teamSize.value || 0) * months * teamImpactPercent;
            const rotationRisk = salaryValue * 2; // Ryzyko rotacji jednej osoby z zespołu
            const teamCostTotal = teamProductivityLoss + rotationRisk;

            // Total
            const total = recruitmentProcessCost + totalSalaryCost + onboardingTotal + 
                         productivityCostTotal + repeatRecruitmentCost + teamCostTotal;

            // Update display
            recruitmentCostEl.textContent = formatCurrency(recruitmentProcessCost);
            salaryCostEl.textContent = formatCurrency(totalSalaryCost);
            onboardingCostEl.textContent = formatCurrency(onboardingTotal);
            productivityCostEl.textContent = formatCurrency(productivityCostTotal);
            repeatCostEl.textContent = formatCurrency(repeatRecruitmentCost);
            teamCostEl.textContent = formatCurrency(teamCostTotal);
            totalCostEl.textContent = formatCurrency(total);

            // Comparisons
            const yearSalary = salaryValue * 12 * (1 + zusEmployer);
            const multiple = (total / yearSalary).toFixed(1);
            salaryMultipleEl.textContent = multiple + '×';
            
            const perMonth = total / months;
            costPerMonthEl.textContent = formatCurrency(perMonth);
        }

        function formatCurrency(value) {
            return new Intl.NumberFormat('pl-PL', {
                style: 'decimal',
                minimumFractionDigits: 0,
                maximumFractionDigits: 0
            }).format(value) + ' zł';
        }
